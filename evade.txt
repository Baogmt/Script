--==================================================
-- DRAG MAIN GUI
--==================================================

local dragging = false
local dragStart
local startPos

Title.InputBegan:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.Touch
	or input.UserInputType == Enum.UserInputType.MouseButton1 then

		dragging = true
		dragStart = input.Position
		startPos = Main.Position
	end
end)

UIS.InputChanged:Connect(function(input)
	if dragging then
		if input.UserInputType == Enum.UserInputType.Touch
		or input.UserInputType == Enum.UserInputType.MouseMovement then

			local delta = input.Position - dragStart

			Main.Position = UDim2.new(
				startPos.X.Scale,
				startPos.X.Offset + delta.X,
				startPos.Y.Scale,
				startPos.Y.Offset + delta.Y
			)
		end
	end
end)

UIS.InputEnded:Connect(function(input)
	if input.UserInputType == Enum.UserInputType.Touch
	or input.UserInputType == Enum.UserInputType.MouseButton1 then
		dragging = false
	end
end)

--==================================================
-- FIND ROBLOX MOBILE JUMP BUTTON
--==================================================

local function GetJumpButton()
	local TouchGui = PlayerGui:FindFirstChild("TouchGui")

	if not TouchGui then
		return nil
	end

	local TouchControlFrame = TouchGui:FindFirstChild("TouchControlFrame")

	if not TouchControlFrame then
		return nil
	end

	local JumpButton = TouchControlFrame:FindFirstChild("JumpButton")

	return JumpButton
end

--==================================================
-- MOVE JUMP BUTTON
--==================================================

local MovingJump = false
local JumpStart
local JumpPosition

JumpMove.MouseButton1Click:Connect(function()

	local JumpButton = GetJumpButton()

	if not JumpButton then
		SensLabel.Text = "Không tìm thấy nút nhảy"
		return
	end

	MovingJump = true

	JumpPosition = JumpButton.Position

	SensLabel.Text = "Kéo nút nhảy trên màn hình"

end)

UIS.InputChanged:Connect(function(input)

	if not MovingJump or Locked then
		return
	end

	if input.UserInputType == Enum.UserInputType.Touch
	or input.UserInputType == Enum.UserInputType.MouseMovement then

		local JumpButton = GetJumpButton()

		if JumpButton then

			local mousePos = input.Position

			local screen = workspace.CurrentCamera.ViewportSize

			local x = mousePos.X / screen.X
			local y = mousePos.Y / screen.Y

			JumpButton.Position = UDim2.new(
				x,
				-JumpButton.AbsoluteSize.X / 2,
				y,
				-JumpButton.AbsoluteSize.Y / 2
			)

		end
	end
end)

UIS.InputEnded:Connect(function(input)

	if input.UserInputType == Enum.UserInputType.Touch
	or input.UserInputType == Enum.UserInputType.MouseButton1 then

		if MovingJump then
			MovingJump = false
			SensLabel.Text = "Độ nhạy: "..Sensitivity
		end

	end

end)

--==================================================
-- LOCK
--==================================================

LockButton.MouseButton1Click:Connect(function()

	Locked = not Locked

	if Locked then
		LockButton.Text = "🔒 Đã khóa"
	else
		LockButton.Text = "🔓 Đang mở khóa"
	end

end)
