local Rayfield = loadstring(game:HttpGet('https://sirius.menu/rayfield'))()

local Window = Rayfield:CreateWindow({
   Name = "⚡ Super Fix Lag & Utilities",
   LoadingTitle = "Đang tải Script...",
   LoadingSubtitle = "by Roblox Studio",
   ConfigurationSaving = { Enabled = false },
   Discord = { Enabled = false },
   KeySystem = false
})

-- Tạo Tab Main chứa tất cả chức năng
local MainTab = Window:CreateTab("Main", 4483362458)

-------------------------------------------------
-- MỤC 1: CÁC NÚT FIX LAG
-------------------------------------------------
MainTab:CreateSection("Tối Ưu Đồ Họa (Fix Lag)")

-- Nút 10%
MainTab:CreateButton({
   Name = "🔧 Fix Lag 10% (Chỉ tắt bóng đổ)",
   Callback = function()
       game:GetService("Lighting").GlobalShadows = false
       Rayfield:Notify({ Title = "10%", Content = "Đã tắt bóng đổ.", Duration = 2 })
   end,
})

-- Nút 50%
MainTab:CreateButton({
   Name = "⚙️ Fix Lag 50% (Tắt hiệu ứng & hạt)",
   Callback = function()
       game:GetService("Lighting").GlobalShadows = false
       for _, v in pairs(workspace:GetDescendants()) do
           if v:IsA("ParticleEmitter") or v:IsA("Smoke") or v:IsA("Fire") or v:IsA("Sparkles") then
               v.Enabled = false
           end
       end
       Rayfield:Notify({ Title = "50%", Content = "Đã tắt hiệu ứng và hạt.", Duration = 2 })
   end,
})

-- Nút 99%
MainTab:CreateButton({
   Name = "🚀 Fix Lag 99% (Tối ưu toàn diện - Nhựa hóa)",
   Callback = function()
       local Lighting = game:GetService("Lighting")
       Lighting.GlobalShadows = false
       Lighting.FogEnd = 9e9
       for _, v in pairs(Lighting:GetChildren()) do
           if v:IsA("PostEffect") then v.Enabled = false end
       end
       
       for _, v in pairs(workspace:GetDescendants()) do
           if v:IsA("BasePart") and not v:IsA("MeshPart") then
               v.Material = Enum.Material.SmoothPlastic
               v.Reflectance = 0
           elseif v:IsA("Decal") or v:IsA("Texture") then
               v:Destroy()
           elseif v:IsA("ParticleEmitter") or v:IsA("Trail") or v:IsA("Smoke") or v:IsA("Fire") then
               v:Destroy()
           end
       end
       
       Rayfield:Notify({ Title = "99%", Content = "Đã tối ưu hóa tối đa đồ họa.", Duration = 3 })
   end,
})

-------------------------------------------------
-- MỤC 2: TÍNH NĂNG BỔ SUNG (HOLD JUMP)
-------------------------------------------------
MainTab:CreateSection("Tính Năng Hỗ Trợ")

local UserInputService = game:GetService("UserInputService")
local LocalPlayer = game:GetService("Players").LocalPlayer
local HoldJumpEnabled = false
local JumpConnection = nil

MainTab:CreateToggle({
   Name = "🦘 Hold Jump (Giữ nút nhảy để nhảy liên tục)",
   CurrentValue = false,
   Flag = "HoldJumpToggle",
   Callback = function(Value)
       HoldJumpEnabled = Value
       
       if HoldJumpEnabled then
           JumpConnection = UserInputService.JumpRequest:Connect(function()
               if HoldJumpEnabled and LocalPlayer.Character and LocalPlayer.Character:FindFirstChildOfClass("Humanoid") then
                   LocalPlayer.Character:FindFirstChildOfClass("Humanoid"):ChangeState(Enum.HumanoidStateType.Jumping)
               end
           end)
           Rayfield:Notify({ Title = "Hold Jump", Content = "Đã BẬT giữ phím nhảy.", Duration = 2 })
       else
           if JumpConnection then
               JumpConnection:Disconnect()
               JumpConnection = nil
           end
           Rayfield:Notify({ Title = "Hold Jump", Content = "Đã TẮT giữ phím nhảy.", Duration = 2 })
       end
   end,
})
