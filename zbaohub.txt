local function PlayCustomMusic()
   if currentMusicSound then
      currentMusicSound:Stop()
      currentMusicSound:Destroy()
      currentMusicSound = nil
   end
   local music = Instance.new("Sound")
   music.SoundId = "rbxassetid://99152674992699"
   music.Volume = 4.5
   music.Looped = true
   music.Parent = SoundService
   music:Play()
   currentMusicSound = music
   Rayfield:Notify({
      Title = "Music Player",
      Content = "Đang phát nhạc ID: 99152674992699 (Volume 4.5)",
      Duration = 3,
   })
end

local ScreenGui = Instance.new("ScreenGui")

ScreenGui.ResetOnSpawn = false
ScreenGui.Parent = game:GetService("CoreGui")

-- Hiệu ứng cầu vồng mượt khi bấm/vuốt qua nút
local function PlayRainbowEffect(btn)
   task.spawn(function()
      local hue = math.random()
      for i = 1, 15 do
         hue = (hue + 0.1) % 1
         btn.BackgroundColor3 = Color3.fromHSV(hue, 1, 1)
         task.wait(0.03)
      end
      btn.BackgroundColor3 = Color3.fromRGB(35, 35, 35)
   end)
end

local function CreateDraggableButton(name, size, position, callback)
   local btn = Instance.new("TextButton")
   btn.Name = name .. "Button"
   btn.Size = size
   btn.Position = position
   btn.BackgroundColor3 = Color3.fromRGB(35, 35, 35)
   btn.BackgroundTransparency = 0.5 -- Nền trong suốt
   btn.BorderColor3 = Color3.fromRGB(255, 255, 255)
   btn.BorderSizePixel = 1
   btn.TextColor3 = Color3.fromRGB(255, 255, 255)
   btn.TextSize = 15
   btn.Font = Enum.Font.SourceSansBold
   btn.Text = name
   btn.Active = true
   btn.Parent = ScreenGui

   local corner = Instance.new("UICorner")
   corner.CornerRadius = UDim.new(0, 8)
   corner.Parent = btn

   local dragging, dragInput, dragStart, startPos

   btn.InputBegan:Connect(function(input)
      if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
         if isDragMode then
            dragging = true
            dragStart = input.Position
            startPos = btn.Position
            input.Changed:Connect(function()
               if input.UserInputState == Enum.UserInputState.End then
                  dragging = false
               end
            end)
         else
            PlaySwitchSound()
            PlayRainbowEffect(btn)
            if callback then callback() end
         end
      end
   end)

   -- Hỗ trợ vuốt ngón tay qua các phím (Swipe / TouchSwipe)
   btn.TouchTap:Connect(function()
      if not isDragMode then
         PlaySwitchSound()
         PlayRainbowEffect(btn)
         if callback then callback() end
      end
   end)

   btn.InputChanged:Connect(function(input)
      if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
         dragInput = input
      end
   end)

   UserInputService.InputChanged:Connect(function(input)
      if input == dragInput and dragging and isDragMode then
         local delta = input.Position - dragStart
         btn.Position = UDim2.new(startPos.X.Scale, startPos.X.Offset + delta.X, startPos.Y.Scale, startPos.Y.Offset + delta.Y)
      end
   end)

   table.insert(draggableFrames, btn)
   return btn
end
