-- Hàm lưu cấu hình tự động vào file JSON
local function SaveKeypadConfig()
    pcall(function()
        if writefile and HttpService then
            local configData = {
                Shape = CurrentShape,
                Size = BaseButtonSize,
                Positions = {}
            }
            for name, data in pairs(RegisteredKeysByName) do
                local pos = data.Btn.Position
                configData.Positions[name] = {
                    XScale = pos.X.Scale, XOffset = pos.X.Offset,
                    YScale = pos.Y.Scale, YOffset = pos.Y.Offset
                }
            end
            writefile(CONFIG_FILE_NAME, HttpService:JSONEncode(configData))
        end
    end)
end

-- Hàm đọc cấu hình tự động từ file JSON
local function LoadKeypadConfig()
    pcall(function()
        if readfile and isfile and isfile(CONFIG_FILE_NAME) and HttpService then
            local result = HttpService:JSONDecode(readfile(CONFIG_FILE_NAME))
            if result then
                if result.Shape then CurrentShape = result.Shape end
                if result.Size then BaseButtonSize = result.Size end
                if result.Positions then
                    for name, posData in pairs(result.Positions) do
                        if RegisteredKeysByName[name] then
                            RegisteredKeysByName[name].Btn.Position = UDim2.new(
                                posData.XScale or 1, posData.XOffset or 0,
                                posData.YScale or 0.5, posData.YOffset or 0
                            )
                        end
                    end
                end
            end
        end
    end)
end

local function CreateCustomButton(name, text, defaultPos, widthMult, keyCodes, isToggle, isMusicToggle)
    local btn = Instance.new("TextButton")
    btn.Name = name
    btn.Size = UDim2.new(0, BaseButtonSize * widthMult, 0, BaseButtonSize)
    btn.Position = defaultPos
    btn.BackgroundColor3 = Color3.fromRGB(20, 20, 20)
    btn.BackgroundTransparency = 0.4
    btn.BorderColor3 = Color3.fromRGB(0, 170, 255)
    btn.BorderSizePixel = 1
    btn.TextColor3 = Color3.fromRGB(255, 255, 255)
    btn.TextSize = 12
    btn.Font = Enum.Font.SourceSansBold
    btn.Text = text
    btn.Active = true
    btn.Parent = ScreenGui

    local corner = Instance.new("UICorner")
    corner.CornerRadius = UDim.new(0, 6)
    corner.Parent = btn

    local isDown = false
    local toggleState = false
    local isActive = false
    local keys = typeof(keyCodes) == "table" and keyCodes or {keyCodes}

    -- Hiệu ứng cầu vồng khi nhấn phím
    task.spawn(function()
        while true do
            if isActive then
                pcall(function()
                    local hue = (tick() % 4) / 4
                    btn.BackgroundColor3 = Color3.fromHSV(hue, 1, 1)
                    btn.BackgroundTransparency = 0.1
                end)
            end
            task.wait(0.03)
        end
    end)

    local function SetVisualState(active)
        isActive = active
        if not active then
            btn.BackgroundColor3 = Color3.fromRGB(20, 20, 20)
            btn.BackgroundTransparency = 0.4
        end
    end

    if isMusicToggle then
        btn.MouseButton1Click:Connect(function()
            if IsEditingLayout then return end
            PlayClick()
            toggleState = not toggleState
            SetVisualState(toggleState)
            
            if toggleState then
                backgroundMusic:Play()
                Rayfield:Notify({ Title = "Nhạc Nền", Content = "Đã BẬT phát nhạc!", Duration = 2 })
            else
                backgroundMusic:Stop()
                Rayfield:Notify({ Title = "Nhạc Nền", Content = "Đã TẮT phát nhạc!", Duration = 2 })
            end
        end)
    elseif isToggle then
        btn.MouseButton1Click:Connect(function()
            if IsEditingLayout then return end
            PlayClick()
            toggleState = not toggleState
            SetVisualState(toggleState)
            for _, k in ipairs(keys) do
                pcall(function() VirtualInputManager:SendKeyEvent(toggleState, k, false, game) end)
            end
        end)
    else
        btn.InputBegan:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                if IsEditingLayout then return end
                isDown = true
                PlayClick()
                SetVisualState(true)
                for _, k in ipairs(keys) do
                    pcall(function() VirtualInputManager:SendKeyEvent(true, k, false, game) end)
                end
            end
        end)

        btn.InputEnded:Connect(function(input)
            if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
                if isDown then
                    isDown = false
                    SetVisualState(false)
                    for _, k in ipairs(keys) do
                        pcall(function() VirtualInputManager:SendKeyEvent(false, k, false, game) end)
                    end
                end
            end)
        end)
    end

    -- Kéo thả tự động lưu vị trí khi thả tay
    local dragging = false
    local dragInput, dragStart, startPos

    btn.InputBegan:Connect(function(input)
        if IsEditingLayout and (input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch) then
            dragging = true
            dragStart = input.Position
            startPos = btn.Position
            input.Changed:Connect(function()
                if input.UserInputState == Enum.UserInputState.End then
                    if dragging then
                        dragging = false
                        SaveKeypadConfig()
                    end
                end
            end)
        end
    end)

    btn.InputChanged:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
            dragInput = input
        end
    end)

    UserInputService.InputChanged:Connect(function(input)
        if input == dragInput and dragging and IsEditingLayout then
            local delta = input.Position - dragStart
            btn.Position = UDim2.new(
                startPos.X.Scale, startPos.X.Offset + delta.X, 
                startPos.Y.Scale, startPos.Y.Offset + delta.Y
            )
        end
    end)

    local keyData = { Name = name, Btn = btn, Corner = corner, WidthMult = widthMult }
    table.insert(RegisteredKeys, keyData)
    RegisteredKeysByName[name] = keyData
    return keyData
end

local function UpdateButtonStyles()
    for _, data in ipairs(RegisteredKeys) do
        data.Btn.Size = UDim2.new(0, BaseButtonSize * data.WidthMult, 0, BaseButtonSize)
        if CurrentShape == "Square" then
            data.Corner.CornerRadius = UDim.new(0, 0)
        elseif CurrentShape == "Circle" then
            data.Corner.CornerRadius = UDim.new(0.5, 0)
        else
            data.Corner.CornerRadius = UDim.new(0, 6)
        end
    end
end

local HoldJumpEnabled = false
local IsJumpingKeyHeld = false
local HoldJumpHeartbeat = nil
local InputBeganConnection = nil
local InputEndedConnection = nil

MainTab:CreateToggle({
   Name = "🦘 Hold Jump (Tự nhảy lại khi đáp đất)",
   CurrentValue = false,
   Flag = "HoldJumpToggle",
   Callback = function(Value)
       PlayClick()
       HoldJumpEnabled = Value
       
       if HoldJumpEnabled then
           InputBeganConnection = UserInputService.InputBegan:Connect(function(input, gameProcessed)
               if not gameProcessed and (input.KeyCode == Enum.KeyCode.Space or input.UserInputType == Enum.UserInputType.Touch) then
                   IsJumpingKeyHeld = true
               end
           end)
           
           InputEndedConnection = UserInputService.InputEnded:Connect(function(input)
               if input.KeyCode == Enum.KeyCode.Space or input.UserInputType == Enum.UserInputType.Touch then
                   IsJumpingKeyHeld = false
               end
           end)
           
           HoldJumpHeartbeat = RunService.Heartbeat:Connect(function()
               pcall(function()
                   if HoldJumpEnabled and IsJumpingKeyHeld then
                       local char = LocalPlayer.Character
                       if char then
                           local humanoid = char:FindFirstChildOfClass("Humanoid")
                           if humanoid and humanoid.FloorMaterial ~= Enum.Material.Air then
                               humanoid:ChangeState(Enum.HumanoidStateType.Jumping)
                           end
                       end
                   end
               end)
           end)
           Rayfield:Notify({ Title = "Hold Jump", Content = "Đã BẬT Hold Jump.", Duration = 2 })
       else
           IsJumpingKeyHeld = false
           if HoldJumpHeartbeat then HoldJumpHeartbeat:Disconnect() HoldJumpHeartbeat = nil end
           if InputBeganConnection then InputBeganConnection:Disconnect() InputBeganConnection = nil end
           if InputEndedConnection then InputEndedConnection:Disconnect() InputEndedConnection = nil end
           Rayfield:Notify({ Title = "Hold Jump", Content = "Đã TẮT Hold Jump.", Duration = 2 })
       end
   end,
})

MainTab:CreateToggle({
   Name = "🐰 Bunny Hop (Bhop gia tốc di chuyển)",
   CurrentValue = false,
   Flag = "BhopToggle",
   Callback = function(Value)
       PlayClick()
       BhopEnabled = Value
       
       if BhopEnabled then
           BhopConnection = RunService.Heartbeat:Connect(function()
               pcall(function()
                   if BhopEnabled then
                       local char = LocalPlayer.Character
                       if char then
                           local humanoid = char:FindFirstChildOfClass("Humanoid")
                           local hrp = char:FindFirstChild("HumanoidRootPart")
                           
                           if humanoid and hrp then
                               if humanoid.FloorMaterial ~= Enum.Material.Air and humanoid.MoveDirection.Magnitude > 0 then
                                   humanoid:ChangeState(Enum.HumanoidStateType.Jumping)
                               end
                               
                               if humanoid.MoveDirection.Magnitude > 0 then
                                   local moveDir = humanoid.MoveDirection
                                   hrp.AssemblyLinearVelocity = Vector3.new(
                                       moveDir.X * BhopSpeed,
                                       hrp.AssemblyLinearVelocity.Y,
                                       moveDir.Z * BhopSpeed
                                   )
                               end
                           end
                       end
                   end
               end)
           end)
           Rayfield:Notify({ Title = "Bhop", Content = "Đã BẬT Bunny Hop.", Duration = 2 })
       else
           if BhopConnection then
               BhopConnection:Disconnect()
               BhopConnection = nil
           end
           Rayfield:Notify({ Title = "Bhop", Content = "Đã TẮT Bunny Hop.", Duration = 2 })
       end
   end,
})
