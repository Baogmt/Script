               if openButton and openButton:IsA("TextButton") then
                  openButton.Text = "Open Menu Cầu Vồng"
                  openButton.TextColor3 = rainbowColor
               end
            end
         end
      end)
      task.wait(0.05)
   end
end)

-- TAB 1: CHÍNH (INSTANT TWISTED & FIX LAG)
---------------------------------------------------------
local MainTab = Window:CreateTab("Chính", 4483362458)

local function getRootPart()
   local char = LocalPlayer.Character or LocalPlayer.CharacterAdded:Wait()
   return char:WaitForChild("HumanoidRootPart", 5)
end

local function instantTwisted()
   local rootPart = getRootPart()
   if rootPart then
      rootPart.CFrame = rootPart.CFrame * CFrame.new(0, 0, 4)
      VirtualInputManager:SendKeyEvent(true, Enum.KeyCode.Q, false, game)
      task.wait(0.05)
      VirtualInputManager:SendKeyEvent(false, Enum.KeyCode.Q, false, game)
   end
end

MainTab:CreateButton({
   Name = "Instant Twisted (Menu)",
   Callback = function()
      instantTwisted()
   end,
})

local function applyFixLag()
   Lighting.GlobalShadows = false
   Lighting.FogEnd = 9e9
   Lighting.Brightness = 2
   for _, v in ipairs(Lighting:GetChildren()) do
      if v:IsA("PostEffect") or v:IsA("Sky") or v:IsA("Atmosphere") then
         v:Destroy()
      end
   end

   if Terrain then
      Terrain.WaterWaveSize = 0
      Terrain.WaterWaveSpeed = 0
      Terrain.WaterTransparency = 1
      Terrain.WaterReflectance = 0
   end

   for _, v in ipairs(workspace:GetDescendants()) do
      if v:IsA("BasePart") and not v.Parent:FindFirstChild("Humanoid") then
         v.Material = Enum.Material.SmoothPlastic
         v.Reflectance = 0
         if v:IsA("MeshPart") then
            v.TextureID = ""
         end
      elseif v:IsA("ParticleEmitter") or v:IsA("Trail") or v:IsA("Fire") or v:IsA("Smoke") or v:IsA("Sparkles") then
         v:Destroy()
      end
   end

   Rayfield:Notify({
      Title = "Fix Lag Thành Công!",
      Content = "Đã tối ưu hóa đồ họa để tăng FPS.",
      Duration = 3,
      Image = 4483362458,
   })
end

MainTab:CreateButton({
   Name = "Fix Lag (Tăng FPS)",
   Callback = function()
      applyFixLag()
   end,
})

local SkinTab = Window:CreateTab("Skin", 4483362458)

local headlessEnabled = false
local korbloxEnabled = false

-- Hàm Fake Headless: Dịch chuyển đầu xuống dưới đất để tránh kéo dài cổ
local function applyHeadless(char)
   if not char then return end
   local head = char:WaitForChild("Head", 3)
   local rootPart = char:WaitForChild("HumanoidRootPart", 3)
   if head and rootPart then
      if headlessEnabled then
         head.CFrame = rootPart.CFrame + Vector3.new(0, -50, 0)
         head.CanCollide = false
         head.Transparency = 1
         for _, v in ipairs(head:GetChildren()) do
            if v:IsA("Decal") or v:IsA("SpecialMesh") then
               v.Transparency = 1
            end
         end
         local neck = head:FindFirstChild("Neck")
         if neck then neck.Enabled = false end
      else
         local neck = head:FindFirstChild("Neck")
         if neck then neck.Enabled = true end
         head.Transparency = 0
         head.CanCollide = true
         for _, v in ipairs(head:GetChildren()) do
            if v:IsA("Decal") or v:IsA("SpecialMesh") then
               v.Transparency = 0
            end
         end
      end
   end
end

-- Hàm apply Korblox chuẩn R6/R15
local function applyKorblox(char)
   if not char then return end
   local humanoid = char:WaitForChild("Humanoid", 3)
   if not humanoid then return end

   if korbloxEnabled then
      if humanoid.RigType == Enum.HumanoidRigType.R6 then
         local rightLeg = char:WaitForChild("Right Leg", 3)
         if rightLeg then
            rightLeg.Transparency = 1
            if not rightLeg:FindFirstChild("KorbloxMeshR6") then
               local mesh = Instance.new("SpecialMesh")
               mesh.Name = "KorbloxMeshR6"
               mesh.MeshId = "rbxassetid://139607718"
               mesh.TextureId = "rbxassetid://139607673"
               mesh.Scale = Vector3.new(1, 1, 1)
               mesh.Parent = rightLeg
            end
         end
      else
         for _, partName in ipairs({"RightUpperLeg", "RightLowerLeg", "RightFoot"}) do
            local part = char:WaitForChild(partName, 3)
            if part then part.Transparency = 1 end
         end
         local rightLower = char:WaitForChild("RightLowerLeg", 3)
         if rightLower and not rightLower:FindFirstChild("KorbloxMeshR15") then
            local mesh = Instance.new("SpecialMesh")
            mesh.Name = "KorbloxMeshR15"
            mesh.MeshId = "rbxassetid://139607718"
            mesh.TextureId = "rbxassetid://139607673"
            mesh.Scale = Vector3.new(1, 1, 1)
            mesh.Parent = rightLower
         end
      end
   else
      if humanoid.RigType == Enum.HumanoidRigType.R6 then
         local rightLeg = char:FindFirstChild("Right Leg")
         if rightLeg then
            rightLeg.Transparency = 0
            if rightLeg:FindFirstChild("KorbloxMeshR6") then rightLeg.KorbloxMeshR6:Destroy() end
         end
      else
         for _, partName in ipairs({"RightUpperLeg", "RightLowerLeg", "RightFoot"}) do
            local part = char:FindFirstChild(partName)
            if part then part.Transparency = 0 end
         end
         local rightLower = char:FindFirstChild("RightLowerLeg")
         if rightLower and rightLower:FindFirstChild("KorbloxMeshR15") then rightLower.KorbloxMeshR15:Destroy() end
      end
   end
end

-- Lắng nghe hồi sinh để giữ nguyên skin khi chết/reset
local function setupCharacter(char)
   char:WaitForChild("Humanoid", 3)
   task.wait(0.3)
   if headlessEnabled then applyHeadless(char) end
   if korbloxEnabled then applyKorblox(char) end
   
   char.ChildAdded:Connect(function(child)
      if child.Name == "Head" and headlessEnabled then
         task.wait(0.1)
         applyHeadless(char)
      elseif (child.Name == "Right Leg" or child.Name == "RightLowerLeg") and korbloxEnabled then
         task.wait(0.1)
         applyKorblox(char)
      end
   end)
end

if LocalPlayer.Character then
   setupCharacter(LocalPlayer.Character)
end

LocalPlayer.CharacterAdded:Connect(function(char)
   setupCharacter(char)
end)

SkinTab:CreateToggle({
   Name = "Fake Headless (Giữ khi chết/reset)",
   CurrentValue = false,
   Flag = "HeadlessToggle",
   Callback = function(Value)
      headlessEnabled = Value
      local char = LocalPlayer.Character
      if char then applyHeadless(char) end
   end,
})

SkinTab:CreateToggle({
   Name = "Fake Korblox (Giữ khi chết/reset)",
   CurrentValue = false,
   Flag = "KorbloxToggle",
   Callback = function(Value)
      korbloxEnabled = Value
      local char = LocalPlayer.Character
      if char then applyKorblox(char) end
   end,
})

local ScreenGui = Instance.new("ScreenGui")
ScreenGui.Name = "MultiButtonsGui"
ScreenGui.ResetOnSpawn = false

if syn and syn.protect_gui then
   syn.protect_gui(ScreenGui)
   ScreenGui.Parent = game.CoreGui
elseif gethui then
   ScreenGui.Parent = gethui()
else
   ScreenGui.Parent = LocalPlayer:WaitForChild("PlayerGui")
end

local function createHoldButton(name, text, defaultPosition, onHoldStart, onHoldEnd)
   local btn = Instance.new("TextButton")
   btn.Name = name
   btn.Size = UDim2.new(0, currentButtonSize, 0, currentButtonSize)
   btn.Position = defaultPosition
   btn.BackgroundColor3 = Color3.fromRGB(30, 30, 30)
   btn.Text = text
   btn.TextColor3 = Color3.fromRGB(255, 255, 255)
   btn.TextSize = 16
   btn.Font = Enum.Font.SourceSansBold
   btn.Parent = ScreenGui

   local corner = Instance.new("UICorner")
   corner.CornerRadius = UDim.new(0.5, 0)
   corner.Parent = btn

   local stroke = Instance.new("UIStroke")
   stroke.Thickness = 2
   stroke.Parent = btn

   task.spawn(function()
      while true do
         stroke.Color = Color3.fromHSV(tick() % 5 / 5, 1, 1)
         task.wait(0.05)
      end
   end)

   local dragging = false
   local dragStart = Vector2.new(0, 0)
   local startPos = UDim2.new(0, 0, 0, 0)
   local isHolding = false

   btn.InputBegan:Connect(function(input)
      if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
         dragging = true
         dragStart = input.Position
         startPos = btn.Position
         
         isHolding = true
         if onHoldStart then onHoldStart() end

         input.Changed:Connect(function()
            if input.UserInputState == Enum.UserInputState.End then
               dragging = false
               if isHolding then
                  isHolding = false
                  if onHoldEnd then onHoldEnd() end
               end
            end
         end)
      end
   end)

   UserInputService.InputChanged:Connect(function(input)
      if dragging and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
         local delta = input.Position - dragStart
         btn.Position = UDim2.new(
            startPos.X.Scale, 
            startPos.X.Offset + delta.X, 
            startPos.Y.Scale, 
            startPos.Y.Offset + delta.Y
         )
      end
   end)

   UserInputService.InputEnded:Connect(function(input)
      if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
         if isHolding then
            isHolding = false
            dragging = false
            if onHoldEnd then onHoldEnd() end
         end
      end
   end)

   return btn
end
