--[[
    ========================================================
    ROBLOX ULTIMATE FPS BOOSTER & LAG REMOVER v3.5
    Tối ưu hóa tối đa - Giảm lag 99% - Tăng FPS ổn định
    ========================================================
]]--

-- Kiểm tra xem game đã tải hoàn tất chưa
if not game:IsLoaded() then
    game.Loaded:Wait()
end

local Players = game:GetService("Players")
local LocalPlayer = Players.LocalPlayer
local Lighting = game:GetService("Lighting")
local Workspace = game:GetService("Workspace")
local Terrain = Workspace:FindFirstChildOfClass("Terrain")

-- Thông báo bắt đầu
print("[FPS BOOSTER]: Đang tiến hành tối ưu hóa game...")

-- 1. Tối ưu hóa Lighting (Ánh sáng & Đồ họa)
pcall(function()
    Lighting.GlobalShadows = false
    Lighting.FogEnd = 9e9
    Lighting.Brightness = 1
    Lighting.ClockTime = 12 -- Cố định trời sáng để giảm tải xử lý bóng đổ thời gian thực
    
    for _, v in ipairs(Lighting:GetChildren()) do
        if v:IsA("PostEffect") or v:IsA("Sky") or v:IsA("Atmosphere") or v:IsA("BlurEffect") or v:IsA("SunRaysEffect") then
            v.Enabled = false
        end
    end
end)

-- 2. Tối ưu hóa Terrain (Địa hình)
pcall(function()
    if Terrain then
        Terrain.WaterWaveSize = 0
        Terrain.WaterWaveSpeed = 0
        Terrain.WaterTransparency = 1
        Terrain.WaterReflectance = 0
        sethiddenproperty(Terrain, "Decoration", false)
    end
end)

-- 3. Xóa/Giảm chi tiết vật liệu và hiệu ứng của toàn bộ Map
pcall(function()
    settings().Rendering.QualityLevel = Enum.QualityLevel.Level01
    
    for _, v in ipairs(Workspace:GetDescendants()) do
        if v:IsA("BasePart") then
            v.Material = Enum.Material.SmoothPlastic
            v.Reflectance = 0
            v.CastShadow = false
        elseif v:IsA("ParticleEmitter") or v:IsA("Trail") or v:IsA("Beam") or v:IsA("Fire") or v:IsA("Smoke") or v:IsA("Sparkles") then
            v.Enabled = false -- Tắt toàn bộ hiệu ứng khói, lửa, vệt sáng gây lag máy yếu
        elseif v:IsA("Decal") or v:IsA("Texture") then
            v.Transparency = 1 -- Ẩn các texture thừa trên tường/vật thể
        end
    end
end)

-- 4. Tự động dọn dẹp rác (Garbage Collection) định kỳ để chống tụt FPS (Drop FPS)
task.spawn(function()
    while task.wait(5) do
        pcall(function()
            collectgarbage("collect")
            game:GetService("RunService"):Set3dRenderingEnabled(true)
        end)
    end
end)

-- 5. Xóa các nhân vật/vật thể không cần thiết ở xa (Low Detail Distance)
pcall(function()
    Workspace.DescendantAdded:Connect(function(v)
        if v:IsA("ParticleEmitter") or v:IsA("Trail") or v:IsA("Beam") then
            v.Enabled = false
        elseif v:IsA("BasePart") then
            v.Material = Enum.Material.SmoothPlastic
            v.CastShadow = false
        end
    end)
end)

-- Thông báo hoàn tất
print("[FPS BOOSTER]: Đã tối ưu hóa thành công! Chúc bạn chơi game mượt mà.")

-- Gửi thông báo nhỏ trong khung chat game (nếu muốn)
pcall(function()
    game:GetService("StarterGui"):SetCore("SendNotification", {
        Title = "FPS Booster Pro",
        Text = "Đã tối ưu hóa giảm lag 99% thành công!",
        Duration = 5
    })
end)
