         end

            if gui.Name == "Rayfield" then
               local openButton = gui:FindFirstChild("OpenButton", true)
               if openButton and openButton:IsA("TextButton") then
                  openButton.Text = "Open Menu Cầu Vồng"
                  openButton.TextColor3 = rainbowColor
               end
            end
         end
      end)
      task.wait(0.05)
   end
end)

---------------------------------------------------------
-- TAB 1: CHÍNH (INSTANT TWISTED & TARGET LOCK)
---------------------------------------------------------
local MainTab = Window:CreateTab("Chính", 4483362458)

local function getRootPart()
   local char = LocalPlayer.Character or LocalPlayer.CharacterAdded:Wait()
   return char:WaitForChild("HumanoidRootPart", 5)
end

local function instantTwisted()
   local rootPart = getRootPart()
   if rootPart then
      rootPart.CFrame = rootPart.CFrame * CFrame.new(0, 0, 4)
      VirtualInputManager:SendKeyEvent(true, Enum.KeyCode.Q, false, game)
      task.wait(0.05)
      VirtualInputManager:SendKeyEvent(false, Enum.KeyCode.Q, false, game)
   end
end

MainTab:CreateButton({
   Name = "Instant Twisted",
   Callback = function()
      instantTwisted()
   end,
})

-- Biến quản lý Target Lock
local targetLockEnabled = false
local lockedTarget = nil

local function getNearestPlayer()
   local shortestDistance = math.huge
   local nearestPlayer = nil
   local myRoot = getRootPart()
   
   if not myRoot then return nil end

   for _, player in ipairs(Players:GetPlayers()) do
      if player ~= LocalPlayer and player.Character then
         local enemyRoot = player.Character:FindFirstChild("HumanoidRootPart")
         local enemyHumanoid = player.Character:FindFirstChild("Humanoid")
         if enemyRoot and enemyHumanoid and enemyHumanoid.Health > 0 then
            local distance = (myRoot.Position - enemyRoot.Position).Magnitude
            if distance < shortestDistance then
               shortestDistance = distance
               nearestPlayer = player
            end
         end
      end
   end
   return nearestPlayer
end

-- Toggle Bật/Tắt hệ thống Target Lock trong Menu
MainTab:CreateToggle({
   Name = "Bật/Tắt Target Lock",
   CurrentValue = false,
   Flag = "TargetLockToggle",
   Callback = function(Value)
      targetLockEnabled = Value
      if not Value then
         lockedTarget = nil
      end
   end,
})

-- Vòng lặp camera khóa vào Torso của mục tiêu gần nhất khi được kích hoạt
RunService.RenderStepped:Connect(function()
   if targetLockEnabled and lockedTarget and lockedTarget.Character then
      local torso = lockedTarget.Character:FindFirstChild("Torso") or lockedTarget.Character:FindFirstChild("UpperTorso")
      local humanoid = lockedTarget.Character:FindFirstChild("Humanoid")
      if torso and humanoid and humanoid.Health > 0 then
         Camera.CFrame = CFrame.new(Camera.CFrame.Position, torso.Position)
      else
         lockedTarget = nil -- Tự động hủy khóa nếu mục tiêu chết hoặc mất phần thân
      end
   end
end)

-- TAB 2: SKIN (FAKE HEADLESS & KORBLOX)
---------------------------------------------------------
local SkinTab = Window:CreateTab("Skin", 4483362458)

local headlessEnabled = false
local korbloxEnabled = false

local function applyHeadless(char)
   if not char then return end
   local head = char:FindFirstChild("Head")
   if head then
      if headlessEnabled then
         head.Transparency = 1
         if head:FindFirstChildOfClass("Decal") then
            head:FindFirstChildOfClass("Decal").Transparency = 1
         end
      else
         head.Transparency = 0
         if head:FindFirstChildOfClass("Decal") then
            head:FindFirstChildOfClass("Decal").Transparency = 0
         end
      end
   end
end

local function applyKorblox(char)
   if not char then return end
   local rightLeg = char:FindFirstChild("Right Leg") or char:FindFirstChild("RightLowerLeg")
   if rightLeg then
      if korbloxEnabled then
         if char:FindFirstChild("Right Leg") then
            char["Right Leg"].Transparency = 1
         else
            if char:FindFirstChild("RightUpperLeg") then char.RightUpperLeg.Transparency = 1 end
            if char:FindFirstChild("RightLowerLeg") then char.RightLowerLeg.Transparency = 1 end
            if char:FindFirstChild("RightFoot") then char.RightFoot.Transparency = 1 end
         end
         
         if not char:FindFirstChild("FakeKorblox") then
            local korbloxPart = Instance.new("SpecialMesh")
            korbloxPart.Name = "KorbloxMesh"
            korbloxPart.MeshId = "rbxassetid://139607718"
            korbloxPart.TextureId = "rbxassetid://139607673"
            korbloxPart.Scale = Vector3.new(1, 1, 1)
            
            local targetPart = char:FindFirstChild("Right Leg") or char:FindFirstChild("RightLowerLeg")
            if targetPart then
               korbloxPart.Parent = targetPart
            end
         end
      else
         if char:FindFirstChild("Right Leg") then
            char["Right Leg"].Transparency = 0
         else
            if char:FindFirstChild("RightUpperLeg") then char.RightUpperLeg.Transparency = 0 end
            if char:FindFirstChild("RightLowerLeg") then char.RightLowerLeg.Transparency = 0 end
            if char:FindFirstChild("RightFoot") then char.RightFoot.Transparency = 0 end
         end
         
         local targetPart = char:FindFirstChild("Right Leg") or char:FindFirstChild("RightLowerLeg")
         if targetPart and targetPart:FindFirstChild("KorbloxMesh") then
            targetPart.KorbloxMesh:Destroy()
         end
      end
   end
end

SkinTab:CreateToggle({
   Name = "Fake Headless",
   CurrentValue = false,
   Flag = "HeadlessToggle",
   Callback = function(Value)
      headlessEnabled = Value
      local char = LocalPlayer.Character
      if char then applyHeadless(char) end
   end,
})

SkinTab:CreateToggle({
   Name = "Fake Korblox (Right Leg)",
   CurrentValue = false,
   Flag = "KorbloxToggle",
   Callback = function(Value)
      korbloxEnabled = Value
      local char = LocalPlayer.Character
      if char then applyKorblox(char) end
   end,
})

LocalPlayer.CharacterAdded:Connect(function(char)
   char:WaitForChild("Humanoid")
   task.wait(0.5)
   if headlessEnabled then applyHeadless(char) end
   if korbloxEnabled then applyKorblox(char) end
end)

end

local FloatingBtn = Instance.new("TextButton")
FloatingBtn.Name = "ActionBtn"
FloatingBtn.Size = UDim2.new(0, 45, 0, 45)
FloatingBtn.Position = UDim2.new(0.1, 0, 0.2, 0)
FloatingBtn.BackgroundColor3 = Color3.fromRGB(30, 30, 30)
FloatingBtn.Text = "Q"
FloatingBtn.TextColor3 = Color3.fromRGB(255, 255, 255)
FloatingBtn.TextSize = 18
FloatingBtn.Font = Enum.Font.SourceSansBold
FloatingBtn.Parent = ScreenGui

local UICorner = Instance.new("UICorner")
UICorner.CornerRadius = UDim.new(0.5, 0)
UICorner.Parent = FloatingBtn

local UIStroke = Instance.new("UIStroke")
UIStroke.Thickness = 2.5
UIStroke.Parent = FloatingBtn

task.spawn(function()
   while true do
      UIStroke.Color = Color3.fromHSV(tick() % 5 / 5, 1, 1)
      task.wait(0.05)
   end
end)

-- Kéo thả nút Q
local draggingQ, dragInputQ, dragStartQ, startPosQ
FloatingBtn.InputBegan:Connect(function(input)
   if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
      draggingQ = true
      dragStartQ = input.Position
      startPosQ = FloatingBtn.Position
      input.Changed:Connect(function()
         if input.UserInputState == Enum.UserInputState.End then draggingQ = false end
      end)
   end
end)
FloatingBtn.InputChanged:Connect(function(input)
   if input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch then
      dragInputQ = input
   end
end)
UserInputService.InputChanged:Connect(function(input)
   if input == dragInputQ and draggingQ then
      local delta = input.Position - dragStartQ
      FloatingBtn.Position = UDim2.new(startPosQ.X.Scale, startPosQ.X.Offset + delta.X, startPosQ.Y.Scale, startPosQ.Y.Offset + delta.Y)
   end
end)

FloatingBtn.MouseButton1Click:Connect(function()
   instantTwisted()
end)

         lockedTarget = nearest
