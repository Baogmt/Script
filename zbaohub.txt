-- Quản lý danh sách các phím & Trạng thái Vuốt (Drag/Swipe)
local RegisteredKeys = {}
local ActiveKey = nil
local CurrentShape = "Rounded" -- Default: Rounded, Square, Circle
local BaseButtonSize = 35

local function TriggerKeyDown(keyData)
    if ActiveKey ~= keyData then
        if ActiveKey then
            ActiveKey.Btn.BackgroundColor3 = Color3.fromRGB(40, 40, 40)
            VirtualInputManager:SendKeyEvent(false, ActiveKey.KeyCode, false, game)
        end
        ActiveKey = keyData
        keyData.Btn.BackgroundColor3 = Color3.fromRGB(0, 170, 255)
        PlayClick()
        VirtualInputManager:SendKeyEvent(true, keyData.KeyCode, false, game)
    end
end

local function TriggerKeyUp()
    if ActiveKey then
        ActiveKey.Btn.BackgroundColor3 = Color3.fromRGB(40, 40, 40)
        VirtualInputManager:SendKeyEvent(false, ActiveKey.KeyCode, false, game)
        ActiveKey = nil
    end
end

-- Xử lý sự kiện di chuyển ngón tay / chuột (Swipe mượt giữa các nút)
local UserInputService = game:GetService("UserInputService")
local isPressing = false

UserInputService.InputBegan:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        isPressing = true
    end
end)

UserInputService.InputEnded:Connect(function(input)
    if input.UserInputType == Enum.UserInputType.MouseButton1 or input.UserInputType == Enum.UserInputType.Touch then
        isPressing = false
        TriggerKeyUp()
    end
end)

UserInputService.InputChanged:Connect(function(input)
    if isPressing and (input.UserInputType == Enum.UserInputType.MouseMovement or input.UserInputType == Enum.UserInputType.Touch) then
        local pos = input.Position
        local found = false
        for _, keyData in ipairs(RegisteredKeys) do
            local btn = keyData.Btn
            local absPos = btn.AbsolutePosition
            local absSize = btn.AbsoluteSize
            if pos.X >= absPos.X and pos.X <= absPos.X + absSize.X and pos.Y >= absPos.Y and pos.Y <= absPos.Y + absSize.Y then
                TriggerKeyDown(keyData)
                found = true
                break
            end
        end
        if not found and ActiveKey then
            TriggerKeyUp()
        end
    end
end)

-- Hàm tạo phím ảo
local function CreateVirtualKey(name, text, gridX, gridY, widthMult, keyCode)
    local btn = Instance.new("TextButton")
    btn.Name = name
    btn.BackgroundColor3 = Color3.fromRGB(40, 40, 40)
    btn.BorderColor3 = Color3.fromRGB(0, 170, 255)
    btn.BorderSizePixel = 1
    btn.TextColor3 = Color3.fromRGB(255, 255, 255)
    btn.TextSize = 13
    btn.Font = Enum.Font.SourceSansBold
    btn.Text = text
    btn.Parent = KeypadFrame

    local corner = Instance.new("UICorner")
    corner.Name = "Corner"
    corner.CornerRadius = UDim.new(0, 6)
    corner.Parent = btn

    local keyData = {
        Btn = btn,
        Corner = corner,
        KeyCode = keyCode,
        GridX = gridX,
        GridY = gridY,
        WidthMult = widthMult or 1
    }

    table.insert(RegisteredKeys, keyData)

    btn.MouseButton1Down:Connect(function()
        TriggerKeyDown(keyData)
    end)

    return keyData
end

-- Cập nhật kích thước & hình dáng nút
local function UpdateKeypadLayout()
    for _, keyData in ipairs(RegisteredKeys) do
        local unit = BaseButtonSize
        local spacing = 4
        
        local w = (unit * keyData.WidthMult) + (spacing * (keyData.WidthMult - 1))
        local h = unit
        local x = keyData.GridX * (unit + spacing)
        local y = keyData.GridY * (unit + spacing)
        
        keyData.Btn.Size = UDim2.new(0, w, 0, h)
        keyData.Btn.Position = UDim2.new(0, x, 0, y)

        if CurrentShape == "Square" then
            keyData.Corner.CornerRadius = UDim.new(0, 0)
        elseif CurrentShape == "Circle" then
            keyData.Corner.CornerRadius = UDim.new(0.5, 0)
        else -- Rounded
            keyData.Corner.CornerRadius = UDim.new(0, 6)
        end
    end
end

MainTab:CreateSection("Tính Năng Di Chuyển")

local HoldJumpEnabled = false
local IsJumpingKeyHeld = false
local HoldJumpHeartbeat = nil
local InputBeganConnection = nil
local InputEndedConnection = nil

MainTab:CreateToggle({
   Name = "🦘 Hold Jump (Tự nhảy lại khi đáp đất)",
   CurrentValue = false,
   Flag = "HoldJumpToggle",
   Callback = function(Value)
       PlayClick()
       HoldJumpEnabled = Value
       
       if HoldJumpEnabled then
           InputBeganConnection = UserInputService.InputBegan:Connect(function(input, gameProcessed)
               if not gameProcessed and (input.KeyCode == Enum.KeyCode.Space or input.UserInputType == Enum.UserInputType.Touch) then
                   IsJumpingKeyHeld = true
               end
           end)
           
           InputEndedConnection = UserInputService.InputEnded:Connect(function(input)
               if input.KeyCode == Enum.KeyCode.Space or input.UserInputType == Enum.UserInputType.Touch then
                   IsJumpingKeyHeld = false
               end
           end)
           
           HoldJumpHeartbeat = RunService.Heartbeat:Connect(function()
               if HoldJumpEnabled and IsJumpingKeyHeld then
                   local char = LocalPlayer.Character
                   if char then
                       local humanoid = char:FindFirstChildOfClass("Humanoid")
                       if humanoid and humanoid.FloorMaterial ~= Enum.Material.Air then
                           humanoid:ChangeState(Enum.HumanoidStateType.Jumping)
                       end
                   end
               end
           end)
           Rayfield:Notify({ Title = "Hold Jump", Content = "Đã BẬT Hold Jump.", Duration = 2 })
       else
           IsJumpingKeyHeld = false
           if HoldJumpHeartbeat then HoldJumpHeartbeat:Disconnect() HoldJumpHeartbeat = nil end
           if InputBeganConnection then InputBeganConnection:Disconnect() InputBeganConnection = nil end
           if InputEndedConnection then InputEndedConnection:Disconnect() InputEndedConnection = nil end
           Rayfield:Notify({ Title = "Hold Jump", Content = "Đã TẮT Hold Jump.", Duration = 2 })
       end
   end,
})

local BhopEnabled = false
local BhopSpeed = 24
local BhopConnection = nil

MainTab:CreateToggle({
   Name = "🐰 Bunny Hop (Bhop gia tốc di chuyển)",
   CurrentValue = false,
   Flag = "BhopToggle",
   Callback = function(Value)
       PlayClick()
       BhopEnabled = Value
       
       if BhopEnabled then
           BhopConnection = RunService.Heartbeat:Connect(function()
               if BhopEnabled then
                   local char = LocalPlayer.Character
                   if char then
                       local humanoid = char:FindFirstChildOfClass("Humanoid")
                       local hrp = char:FindFirstChild("HumanoidRootPart")
                       
                       if humanoid and hrp then
                           if humanoid.FloorMaterial ~= Enum.Material.Air and humanoid.MoveDirection.Magnitude > 0 then
                               humanoid:ChangeState(Enum.HumanoidStateType.Jumping)
                           end
                           
                           if humanoid.MoveDirection.Magnitude > 0 then
                               local moveDir = humanoid.MoveDirection
                               hrp.AssemblyLinearVelocity = Vector3.new(
                                   moveDir.X * BhopSpeed,
                                   hrp.AssemblyLinearVelocity.Y,
                                   moveDir.Z * BhopSpeed
                               )
                           end
                       end
                   end
               end
           end)
           Rayfield:Notify({ Title = "Bhop", Content = "Đã BẬT Bunny Hop.", Duration = 2 })
       else
           if BhopConnection then
               BhopConnection:Disconnect()
               BhopConnection = nil
           end
           Rayfield:Notify({ Title = "Bhop", Content = "Đã TẮT Bunny Hop.", Duration = 2 })
       end
   end,
})
