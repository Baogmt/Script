-- Tải thư viện Rayfield UI
local Rayfield = loadstring(game:HttpGet('https://sirius.menu/rayfield'))()

-- Tạo Cửa sổ (Window)
local Window = Rayfield:CreateWindow({
   Name = "Footstep Sound - Ultra Volume",
   LoadingTitle = "Đang tải script...",
   ConfigurationSaving = { Enabled = false },
   Discord = { Enabled = false },
   KeySystem = false
})

local MainTab = Window:CreateTab("Cấu Hình Âm Thanh", 4483362458)

-- Biến cấu hình
local soundEnabled = false
local soundSource = "https://files.catbox.moe/ixpw3t.mp3"
local soundVolume = 10 -- Mặc định để to
local stepCooldown = 0.35
local boostMultiplier = 3 -- Số lượng kênh âm thanh phát cùng lúc để kích to tiếng

-- Tạo sẵn danh sách các Sound Object để phát cộng hưởng
local soundPool = {}
local SoundService = game:GetService("SoundService")

local function createSoundPool()
    -- Xóa pool cũ nếu có
    for _, sound in ipairs(soundPool) do
        sound:Destroy()
    end
    soundPool = {}
    
    -- Tạo mới các kênh phát âm thanh
    for i = 1, boostMultiplier do
        local snd = Instance.new("Sound")
        snd.Name = "BoostedFootstep_" .. i
        snd.SoundId = soundSource
        snd.Volume = soundVolume
        snd.Parent = SoundService
        table.insert(soundPool, snd)
    end
end

createSoundPool()

-- Logic phát âm thanh khi di chuyển
local Players = game:GetService("Players")
local LocalPlayer = Players.LocalPlayer
local lastStepTime = 0

game:GetService("RunService").Heartbeat:Connect(function()
    if not soundEnabled then return end
    
    local character = LocalPlayer.Character
    if character and character:FindFirstChild("Humanoid") and character:FindFirstChild("HumanoidRootPart") then
        local humanoid = character.Humanoid
        local rootPart = character.HumanoidRootPart
        
        local isMoving = rootPart.Velocity.Magnitude > 2 and humanoid.MoveDirection.Magnitude > 0
        local isGrounded = humanoid.FloorMaterial ~= Enum.Material.Air
        
        if isMoving and isGrounded then
            local currentTime = tick()
            if currentTime - lastStepTime >= stepCooldown then
                lastStepTime = currentTime
                
                -- Phát đồng thời tất cả các kênh tiếng trong Pool để kích âm lượng cực đại
                for _, snd in ipairs(soundPool) do
                    snd.SoundId = soundSource
                    snd.Volume = soundVolume
                    snd:Play()
                end
            end
        end
    end
end)

--- GIAO DIỆN RAYFIELD ---

MainTab:CreateToggle({
   Name = "Bật Âm Thanh Bước Chân",
   CurrentValue = false,
   Flag = "FootstepToggle",
   Callback = function(Value)
       soundEnabled = Value
   end,
})

MainTab:CreateSlider({
   Name = "Âm Lượng (Volume)",
   Range = {0, 50}, -- Cho phép chỉnh lên mức 50 (mặc định Roblox chỉ tới 10)
   Increment = 1,
   Suffix = "x",
   CurrentValue = 10,
   Flag = "VolumeSlider",
   Callback = function(Value) 
       soundVolume = Value
       for _, snd in ipairs(soundPool) do
           snd.Volume = soundVolume
       end
   end,
})

MainTab:CreateSlider({
   Name = "Độ Kích Âm (Chèn Thêm Kênh Tiếng)",
   Range = {1, 5}, -- Phát từ 1 đến 5 lớp âm thanh cùng lúc
   Increment = 1,
   Suffix = " lớp tiếng",
   CurrentValue = 3,
   Flag = "BoostSlider",
   Callback = function(Value)
       boostMultiplier = Value
       createSoundPool()
   end,
})

MainTab:CreateSlider({
   Name = "Tốc Độ Nhịp Bước (Cooldown)",
   Range = {0.1, 1},
   Increment = 0.05,
   Suffix = "s",
   CurrentValue = 0.35,
   Flag = "CooldownSlider",
   Callback = function(Value) stepCooldown = Value end,
})
